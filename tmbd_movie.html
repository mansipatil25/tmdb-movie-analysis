df['revenue'] = pd.to_numeric(df['revenue'], errors='coerce')
df['budget'] = pd.to_numeric(df['budget'], errors='coerce')

# Filter only valid positive values
df_clean = df[(df['revenue'] > 0) & (df['budget'] > 0)]

# Scatter plot
plt.figure(figsize=(9,5))
plt.scatter(df_clean['popularity'], df_clean['revenue'] / 1e6,
            alpha=0.4, color='steelblue', edgecolors='none', s=25)
plt.title('Popularity vs Revenue')
plt.xlabel('Popularity score')
plt.ylabel('Revenue ($M)')
plt.tight_layout()
plt.show()

# Correlation
corr = df_clean['popularity'].corr(df_clean['revenue'])
print(f"Correlation: {corr:.2f}")

month_names = {1:'Jan',2:'Feb',3:'Mar',4:'Apr',5:'May',6:'Jun',
               7:'Jul',8:'Aug',9:'Sep',10:'Oct',11:'Nov',12:'Dec'}

monthly = df.groupby('release_month')['revenue'].mean() / 1e6
monthly.index = monthly.index.map(month_names)

monthly.plot(kind='bar', color='coral', figsize=(10,5))
plt.title('Average revenue by release month ($M)')
plt.xlabel('Month')
plt.ylabel('Avg revenue ($M)')
plt.xticks(rotation=0)
plt.tight_layout()
plt.show()

print("Best month:", monthly.idxmax(), "-", round(monthly.max(),1), "M avg")
